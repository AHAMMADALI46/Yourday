function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var folderName = data.folderName || 'Yourday Journals';
  var folders = DriveApp.getFoldersByName(folderName);
  var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);
  upsertFile(folder, 'yourday-vault-backup.json', JSON.stringify(data.vault, null, 2), MimeType.PLAIN_TEXT);
  (data.markdownFiles || []).forEach(function(f) {
    upsertFile(folder, f.fileName, f.markdown, MimeType.PLAIN_TEXT);
  });
  return ContentService.createTextOutput(JSON.stringify({ ok: true, folderId: folder.getId(), folderUrl: folder.getUrl() }))
    .setMimeType(ContentService.MimeType.JSON);
}
function upsertFile(folder, name, content, mime) {
  var files = folder.getFilesByName(name);
  if (files.hasNext()) { files.next().setContent(content); }
  else { folder.createFile(name, content, mime); }
}
